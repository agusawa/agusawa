# Project Switch Tracker Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a standalone single-page React + Tailwind app (`apps/switch-tracker`) that tracks parallel AI-assisted coding projects in `localStorage`, with a status-driven focus banner, cap warning, and rest timer.

**Architecture:** Vite + React 19 + TypeScript + Tailwind CSS 4, added as a new package to the existing pnpm workspace. `App.tsx` owns all project state and localStorage sync; presentation is split into small single-purpose components (`FocusBanner`, `CapWarning`, `AddProjectForm`, `ProjectCard`, `RestTimer`, `Toast`) that receive data/callbacks as props. Pure derivation logic (sorting rank, banner message, cap check) lives in plain TS modules so it has no React dependency.

**Tech Stack:** Vite 6, React 19, TypeScript 5, Tailwind CSS 4 (`@tailwindcss/vite`), pnpm workspace (`apps/*`), Docker + nginx for deployment (mirrors `apps/til`).

## Global Constraints

- No backend, no accounts — all persistence via browser `localStorage` only.
- Single column layout at all viewport widths, capped at a comfortable `max-w-xl` reading width — no responsive grid.
- Status colors: waiting=amber, review=blue, idle=stone/gray, done=emerald — muted 500/600 shades, not saturated.
- No external icon library — emoji (🟡🔵⚪🟢) only.
- Rest timer notification is an in-page toast only — no browser Notification API.
- No confirmation dialog on delete (low-stakes local data).
- No test framework is being introduced for this app; verification is manual (dev server / build output) as specified per task.
- pnpm workspace package name for this app: `switch-tracker`, directory `apps/switch-tracker`.

```ts
export const STATUS_RANK: Record<Status, number> = STATUS_ORDER.reduce(
```

```ts
import type { Project } from "./types";

const STORAGE_KEY = "switch-tracker.projects";

export function loadProjects(): Project[] {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return [];
    const parsed = JSON.parse(raw);
    if (!Array.isArray(parsed)) return [];
    return parsed as Project[];
  } catch {
    return [];
  }
}

export function saveProjects(projects: Project[]): void {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(projects));
}
```

- [ ] **Step 4: Create `apps/switch-tracker/src/focusLogic.ts`**

```ts
import type { Project } from "./types";

export function getBannerMessage(projects: Project[]): string | null {
  if (projects.length === 0) return null;

  const reviewCount = projects.filter((p) => p.status === "review").length;
  if (reviewCount > 0) {
    return `You have ${reviewCount} project${reviewCount === 1 ? "" : "s"} ready to review — batch these together.`;
  }

  const active = projects.filter((p) => p.status !== "done");
  if (active.length > 0 && active.every((p) => p.status === "waiting")) {
    return "Everything's waiting on AI. Take a 5-minute break before you check back.";
  }

  const waitingCount = projects.filter((p) => p.status === "waiting").length;
  if (waitingCount === 0 && reviewCount === 0) {
    return "Nothing pending — good time to fire off next tasks.";
  }

  return null;
}

export function isOverCap(projects: Project[]): boolean {
  const activeCount = projects.filter(
    (p) => p.status === "waiting" || p.status === "review"
  ).length;
  return activeCount > 2;
}
```

- [ ] **Step 1: Replace `apps/switch-tracker/src/App.tsx` with the stateful shell**

```tsx
import { useEffect, useMemo, useState } from "react";
import type { Project } from "./types";
import { loadProjects, saveProjects } from "./storage";
import { STATUS_RANK } from "./statusConfig";
import { getBannerMessage, isOverCap } from "./focusLogic";

export function App() {
  const [projects, setProjects] = useState<Project[]>(() => loadProjects());

  useEffect(() => {
    saveProjects(projects);
  }, [projects]);

  const sortedProjects = useMemo(
    () => [...projects].sort((a, b) => STATUS_RANK[a.status] - STATUS_RANK[b.status]),
    [projects]
  );

  const bannerMessage = useMemo(() => getBannerMessage(projects), [projects]);
  const overCap = useMemo(() => isOverCap(projects), [projects]);

  function handleAdd(name: string) {
    const project: Project = {
      id: crypto.randomUUID(),
      name,
      status: "idle",
      lastAsked: "",
      expecting: "",
      watchFor: "",
      nextAction: "",
      updatedAt: Date.now(),
    };
    setProjects((prev) => [...prev, project]);
  }

  function handleUpdate(id: string, patch: Partial<Project>) {
    setProjects((prev) =>
      prev.map((p) => (p.id === id ? { ...p, ...patch, updatedAt: Date.now() } : p))
    );
  }

  function handleDelete(id: string) {
    setProjects((prev) => prev.filter((p) => p.id !== id));
  }

  return (
    <div className="mx-auto max-w-xl px-4 py-8">
      <header className="mb-6">
        <h1 className="text-lg font-semibold text-stone-800">Project Switch Tracker</h1>
      </header>

      {projects.length === 0 ? (
        <div className="rounded-lg border border-dashed border-stone-300 px-4 py-10 text-center">
          <p className="text-sm text-stone-500">No projects yet — add one to start tracking.</p>
        </div>
      ) : (
        <div className="flex flex-col gap-4">
          <p className="text-sm text-stone-700">{bannerMessage}</p>
          {overCap && (
            <p className="text-xs text-stone-500">
              3+ active projects — consider dropping to 2 for a bit.
            </p>
          )}
          <ul className="flex flex-col gap-2 text-sm text-stone-700">
            {sortedProjects.map((p) => (
              <li key={p.id}>
                {p.name} — {p.status}
              </li>
            ))}
          </ul>
        </div>
      )}
    </div>
  );
}
```

```tsx
import { useState } from "react";
import type { Project } from "../types";
import { STATUS_CONFIG, STATUS_ORDER } from "../statusConfig";

interface ProjectCardProps {
  project: Project;
  onUpdate: (id: string, patch: Partial<Project>) => void;
  onDelete: (id: string) => void;
}

const EDITABLE_FIELDS: { key: keyof Project; label: string }[] = [
  { key: "lastAsked", label: "Last asked" },
  { key: "expecting", label: "Expecting" },
  { key: "watchFor", label: "Watch for" },
  { key: "nextAction", label: "Next action" },
];

export function ProjectCard({ project, onUpdate, onDelete }: ProjectCardProps) {
  const [expanded, setExpanded] = useState(false);
  const meta = STATUS_CONFIG[project.status];

  return (
    <div className="rounded-lg border border-stone-200 bg-white p-4 shadow-sm">
      <div className="flex items-center justify-between gap-3">
        <button
          type="button"
          onClick={() => setExpanded((v) => !v)}
          className="flex flex-1 items-center gap-2 text-left"
        >
          <span aria-hidden="true">{meta.emoji}</span>
          <span className={`text-xs font-medium ${meta.text}`}>{meta.label}</span>
          <span className="text-sm font-semibold text-stone-800">{project.name}</span>
        </button>

        <div className="flex items-center gap-1">
          {STATUS_ORDER.map((status) => (
            <button
              key={status}
              type="button"
              title={STATUS_CONFIG[status].label}
              onClick={() => onUpdate(project.id, { status })}
              className={`h-3.5 w-3.5 rounded-full ${STATUS_CONFIG[status].dot} ${
                project.status === status ? "ring-2 ring-offset-1 ring-stone-400" : "opacity-40"
              }`}
            />
          ))}
        </div>

        <button
          type="button"
          onClick={() => onDelete(project.id)}
          className="text-xs text-stone-400 hover:text-red-500"
        >
          Delete
        </button>
      </div>

      {expanded && (
        <div className="mt-3 grid gap-2">
          {EDITABLE_FIELDS.map(({ key, label }) => (
            <EditableField
              key={key}
              label={label}
              value={project[key] as string}
              onCommit={(value) => {
                if (value === project[key]) return;
                onUpdate(project.id, { [key]: value } as Partial<Project>);
              }}
            />
          ))}
        </div>
      )}
    </div>
  );
}

interface EditableFieldProps {
  label: string;
  value: string;
  onCommit: (value: string) => void;
}

function EditableField({ label, value, onCommit }: EditableFieldProps) {
  const [draft, setDraft] = useState(value);

  return (
    <label className="flex flex-col gap-1">
      <span className="text-xs font-medium text-stone-500">{label}</span>
      <input
        type="text"
        value={draft}
        onChange={(e) => setDraft(e.target.value)}
        onBlur={() => onCommit(draft)}
        className="rounded-md border border-stone-300 px-2 py-1 text-sm text-stone-800 focus:border-stone-500 focus:outline-none"
      />
    </label>
  );
}
```

- [ ] **Step 1: Create `apps/switch-tracker/src/components/RestTimer.tsx`**

```tsx
import { useEffect, useState } from "react";

const REST_SECONDS = 5 * 60;

interface RestTimerProps {
  onDone: () => void;
}

export function RestTimer({ onDone }: RestTimerProps) {
  const [remaining, setRemaining] = useState<number | null>(null);

  useEffect(() => {
    if (remaining === null) return;
    if (remaining <= 0) {
      setRemaining(null);
      onDone();
      return;
    }
    const timer = setTimeout(() => setRemaining((r) => (r ?? 1) - 1), 1000);
    return () => clearTimeout(timer);
  }, [remaining, onDone]);

  if (remaining !== null) {
    const minutes = Math.floor(remaining / 60);
    const seconds = remaining % 60;
    return (
      <span className="text-sm text-stone-500">
        Stepping away — {minutes}:{seconds.toString().padStart(2, "0")}
      </span>
    );
  }

  return (
    <button
      type="button"
      onClick={() => setRemaining(REST_SECONDS)}
      className="text-sm text-stone-500 underline decoration-dotted hover:text-stone-700"
    >
      Step away — 5 min
    </button>
  );
}
```

```tsx
import { useEffect } from "react";

interface ToastProps {
  message: string | null;
  onDismiss: () => void;
}

export function Toast({ message, onDismiss }: ToastProps) {
  useEffect(() => {
    if (!message) return;
    const timer = setTimeout(onDismiss, 4000);
    return () => clearTimeout(timer);
  }, [message, onDismiss]);

  if (!message) return null;

  return (
    <div className="fixed bottom-4 left-1/2 -translate-x-1/2 rounded-md bg-stone-800 px-4 py-2 text-sm text-white shadow-lg">
      {message}
    </div>
  );
}
```
